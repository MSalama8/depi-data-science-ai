01

الاتصال بقاعدة البيانات — psycopg2.connect()

أول خطوة في أي سكريبت: تفتح قناة اتصال حقيقية بين بايثون وسيرفر PostgreSQL. محتاج 4 حاجات بالظبط: مكان السيرفر، اسم قاعدة البيانات، اسم المستخدم، والباسورد.

connect.py
import psycopg2

try:
    conn = psycopg2.connect(
        host="localhost",
        database="company_db",
        user="postgres",
        password="root"
    )
    print("Connection to the database established successfully.")
except Exception as e:
    print(f"Error connecting to the database: {e}")
▸ فكّر فيها كده
◂host = عنوان السيرفر
◂database = اسم قاعدة البيانات
◂user = اسم المستخدم
◂password = كلمة السر
◂conn هو "الخط" اللي واصل بايثون بالسيرفر — من غيره مفيش أي تعامل ممكن
ليه try/except من أول سطر؟ الاتصال ممكن يفشل لأسباب كتير: السيرفر مقفول، الباسورد غلط، اسم قاعدة البيانات غلط. من غير try/except، أي فشل هيوقف البرنامج كله برسالة تكنيكية مخيفة (traceback) بدل رسالة واضحة.
02

الـ Cursor — إيه هو وليه محتاجينه؟

فكّر في conn إنه خط التليفون الواصل بين بايثون والداتابيز، والـ cursor هو "الإيد" اللي بتتحرك جوه الداتابيز، تنفّذ الأوامر، وترجّعلك النتايج.

cursor.py
cursor = conn.cursor()
خطر شائع: من غير cursor، مينفعش تبعت أي أمر SQL خالص. أي عملية (SELECT, INSERT, UPDATE...) لازم تمر بالـ cursor أولًا.
03

execute() — تنفيذ الاستعلام

بتبعت جملة الـ SQL للسيرفر عشان ينفّذها ويجهّز النتيجة — لكنها مبترجعش النتيجة فعليًا. عشان تجيب النتيجة، محتاج دالة fetch منفصلة (الموضوع الجاي).

execute.py
cursor.execute("select * from employee")
افتكرها كده: execute() = "يا سيرفر، نفّذ الاستعلام ده وجهّز النتيجة عندك". fetch...() = "دلوقتي ابعتلي النتيجة اللي جهّزتها".
04

الفرق الجوهري: fetchone / fetchmany / fetchall

أي استعلام SQL بيرجّع "مجموعة نتائج" (result set)، والـ cursor بيقف عند صف معيّن جواها — وكل دالة بتتعامل مع الموقف ده بطريقة مختلفة.

الدالةبترجع إيهبتحرك المؤشر إد إيه
fetchone()صف واحد بس (اللي المؤشر واقف عنده)خطوة واحدة للصف اللي بعده
fetchmany(n)أول n صف من مكان المؤشرn خطوة
fetchall()كل الصفوف المتبقية دفعة واحدةلحد آخر النتيجة
خطر شائع: fetchall() خطيرة لو الجدول فيه ملايين الصفوف — هتتحمّل كلها في الذاكرة (RAM) دفعة واحدة. لو الجدول كبير، استخدم fetchmany(n) في لوب بدل fetchall().
05

scroll() — تحريك المؤشر من غير قراءة

scroll() مختلفة تمامًا عن دوال fetch — هي مش "بتقرأ" أي حاجة، بس بتحرّك المؤشر لمكان تاني في مجموعة النتائج.

scroll.py
cursor.execute("select * from employee")
cursor.scroll(2)              # انقل المؤشر خطوتين للقدام
rows = cursor.fetchmany(5)   # واقرأ 5 صفوف من المكان الجديد
▸ فكّر فيها كده
◂الصفوف: [0] [1] [2] [3] [4] [5] ...
◂المؤشر بيبدأ عند [0]
◂scroll(2) → المؤشر يتحرك لـ [2] (من غير أي قراءة)
◂fetchmany(5) → يقرا من [2] لحد [6]
06

INSERT وParameterized Query — ليه %s مش f-string؟

إضافة صف واحد، وأهم قاعدة أمان في الموضوع كله: استخدام %s بدل ما تحط القيم مباشرة جوه النص.

insert.py
insert_row = ("12123456", "sakr", "mo", 80000)
insert_query = "insert into employee (ssn, fname, lname, salary) values (%s, %s, %s, %s)"
cursor.execute(insert_query, insert_row)
conn.commit()
ليه مش نكتب f"...values ({ssn}, ...)"؟ ده اسمه Parameterized Query، وبيمنع مشكلة اسمها SQL Injection. لو حد كتب في القيمة حاجة زي '; DROP TABLE employee; --، هتتفسّر كنص عادي آمن، مش كأمر SQL خطير هيتنفّذ فعليًا.
لاحظ: execute() هنا بتاخد باراميتر تاني (الـ tuple)، وpsycopg2 هي اللي بتحط القيم مكان الـ %s بأمان — مش انت اللي بتعمل ده يدويًا.
07

executemany() — إدخال أكتر من صف مرة واحدة

نفس فكرة INSERT، بس بدل ما تكرر execute() في لوب لكل صف، executemany() بتاخد لستة من الـ tuples وتدخلهم كلهم بأمر واحد.

executemany.py
insert_row = [
    ("121234568", "mohamed", "mahmoud", 90000),
    ("12123457", "ahmed", "ali", 90000),
]

cursor.executemany(insert_query, insert_row)
conn.commit()
ليه أسرع؟ بدل ما تبعت للسيرفر أمر منفصل لكل صف (رحلة ذهاب وإياب لكل واحد)، executemany() بتبعت كل الصفوف مرة واحدة — تقليل عدد الرحلات (round trips) بين بايثون والسيرفر.
08

بحث ديناميكي بمدخلات المستخدم

هنا القيمة جاية من المستخدم مباشرة (input()) — وده بالظبط الموقف اللي لازم تستخدم فيه %s إجباريًا، مش اختياريًا.

search.py
ssn = input("Enter the ssn of the employee to fetch details: ")
cursor.execute(
    "select fname, lname, salary from employee where ssn = %s",
    (ssn,)
)
row = cursor.fetchone()
▸ ليه دي حساسة أمنيًا؟
◂القيمة جاية من مستخدم حقيقي — مش انت اللي كاتبها في الكود
◂f"...where ssn = {ssn}" → لو ssn جاله قيمة خبيثة، بتتنفذ كأمر SQL حقيقي. خطر!
◂"...where ssn = %s", (ssn,) → القيمة بتتعامل معاها كنص دايمًا، مهما كان محتواها
لاحظ: (ssn,) — الفاصلة دي إجبارية، عشان بايثون تفهم إنها tuple فيه عنصر واحد، مش مجرد قوسين حوالين قيمة عادية.
09

try/except/rollback — بعمق أكتر

السر في try/except مش بس "امسك الخطأ" — هو بيحدد مسار التنفيذ الفعلي للبرنامج. ليه فيه try منفصل لكل مرحلة؟

insert_full.py
try:                                    # Try #1 — الاتصال
    conn = psycopg2.connect(...)
    print("Connection ... successfully.")
except Exception as e:
    print(f"Error connecting ...: {e}")

cursor = conn.cursor()

try:                                    # Try #2 — الـ insert
    cursor.execute(insert_query, insert_row)
    conn.commit()
    print("Row inserted successfully.")
except Exception as e:
    print(f"Error inserting row: {e}")
    conn.rollback()
finally:
    # conn.close()
    pass
ليه try منفصل لكل مرحلة؟ كل مرحلة عندها مخاطر مختلفة تمامًا — try الاتصال ممكن يفشل بسبب باسورد غلط أو سيرفر مقفول (خطأ "قبل" ما تلمس أي بيانات)، بينما try الـ insert ممكن يفشل بسبب ssn مكرر أو نوع بيانات غلط. لو حطيت كل حاجة في try واحد كبير، مش هتعرف بالظبط فين حصلت المشكلة.
conn.rollback() — أهم سطر في الملف كله: لما تعمل execute() لعملية INSERT، الداتابيز مش بتثبت التغيير فورًا — بتفتح transaction مؤقتة. لو حصل خطأ في نص العملية ومعملتش rollback()، الاتصال هيفضل "عالق" في transaction معلّقة، وأي أمر تاني هتحاول تنفّذه على نفس الـ connection ممكن يرفض يشتغل. افتكرها كده: commit() = احفظ التغيير فعليًا، rollback() = الغِ كل حاجة عملتها من أول العملية دي، ارجع الوضع زي ما كان.
10

ليه except Exception العامة مش أفضل ممارسة؟

استخدام Exception العامة بيمسك أي خطأ ممكن يحصل، حتى لو مش متعلق بالداتابيز خالص. في مشروع حقيقي، الأفضل تستخدم أنواع أخطاء psycopg2 المتخصصة.

specific_errors.py
import psycopg2

try:
    cursor.execute(insert_query, insert_row)
    conn.commit()
except psycopg2.IntegrityError as e:     # مثلاً: ssn مكرر
    print(f"Duplicate or constraint error: {e}")
    conn.rollback()
except psycopg2.OperationalError as e:   # مشكلة في الاتصال نفسه
    print(f"Connection issue: {e}")
except psycopg2.Error as e:              # أي خطأ تاني متعلق بالداتابيز
    print(f"Database error: {e}")
    conn.rollback()
الفايدة العملية: كده لو حصل duplicate ssn، هتعرف بالظبط إن المشكلة "constraint violation" مش "مشكلة اتصال" — وتقدر تتعامل مع كل حالة بطريقة مختلفة تناسبها (زي إنك تطلب من المستخدم ssn تاني بدل ما توقف البرنامج كله).
11

finally — تنظيف الموارد

finally بتتنفّذ دايمًا، سواء العملية نجحت أو فشلت. أفضل استخدام ليها هو قفل الاتصال بالداتابيز مهما حصل.

current_code.py
finally:
    # conn.close()
    pass
better_practice.py
finally:
    if conn:
        conn.close()
        print("Connection closed.")
مشكلة الكود الحالي: conn.close() متعملة comment وحطينا pass بدالها — يعني الاتصال فعليًا بيفضل مفتوح لحد ما البرنامج يقفل. لو الملف اتشغّل مرات كتير (زي في اختبار أو تطوير)، هتتراكم عندك اتصالات مفتوحة بتستهلك موارد السيرفر من غير داعي.
12

القاعدة الذهبية: try / except / finally

ملخّص المسار الكامل لأي عملية على قاعدة البيانات، من الاتصال لحد التنظيف.

الجزءبيتنفّذ إمتىالاستخدام المثالي
tryدايمًا (أول حاجة)الكود اللي فيه احتمال خطأ
exceptبس لو حصل خطأالتعامل مع الخطأ + rollback() لو transaction مفتوحة
finallyدايمًا (نجح أو فشل)تنظيف الموارد — قفل الاتصال
خطة تفكير سريعة لأي سكريبت psycopg2 جديد: اتصل (try/except) → افتح cursor → نفّذ الاستعلام (execute) → لو SELECT: اجلب النتيجة بالدالة المناسبة (fetchone/many/all حسب الحجم) → لو INSERT/UPDATE/DELETE: اعمل commit() لو نجح أو rollback() لو فشل (جوه try/except منفصل) → في الآخر: اقفل الاتصال جوه finally مهما حصل.
