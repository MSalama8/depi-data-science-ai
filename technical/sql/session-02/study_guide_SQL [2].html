01

Comparison Conditions + LIKE / ILIKE

أدوات المقارنة الأساسية (=، <>، >، <، BETWEEN، IN) وأدوات البحث بالنمط (LIKE / ILIKE) — أساس أي شرط WHERE تقريبًا.

الأداةالمعنى
=يساوي بالظبط
<> أو !=لا يساوي
BETWEEN a AND bبين قيمتين (شامل الطرفين)
IN (list)يطابق أي قيمة من لستة
%أي عدد من الحروف (صفر أو أكتر)
_حرف واحد بالظبط
comparison.sql
select * from employee where  salary > 30000;

select fname , lname , salary
from employee
where salary between 20000 and 25000;

select fname , lname , salary , dno
from employee where dno in (10,2,4);
like_ilike.sql
select fname , lname , salary from employee
where fname ilike 'a%';   -- يبدأ بحرف a

select fname , lname , salary from employee
where fname ilike '%d';   -- ينتهي بحرف d

select fname , lname , salary from employee
where fname ilike '_a%';  -- الحرف الثاني a
LIKE مقابل ILIKE: في PostgreSQL، LIKE بتفرّق بين الحروف الكبيرة والصغيرة (case-sensitive)، أما ILIKE فبتتجاهل الفرق — يعني 'Ahmad' و'ahmad' هيتطابقوا مع بعض معاها. غالبًا ILIKE أأمن للاستخدام في البحث النصي.
02

Logical Conditions — AND / OR / NOT

بتخليك تجمع أكتر من شرط مع بعض. AND = لازم الشرطين صح، OR = يكفي شرط واحد صح، NOT = بتعكس الشرط.

and_or.sql
select fname , lname , sex , salary from employee
where salary > 30000 and sex ilike 'M';

select fname , lname , salary from employee
where salary > 50000 or salary < 21000;
parentheses.sql
-- ذكور راتبهم أكبر من 40000 أو إناث راتبهم أكبر من 40000
select fname , lname , salary from employee
where (salary > 40000 and sex ilike 'm')
   or (salary > 40000 and sex ilike 'f');
not_condition.sql
select fname , lname , salary from employee
where not salary > 30000;

-- بيدّي نفس نتيجة:
select fname , lname , salary from employee
where salary < 30000;
▸ فكّر فيها كده
NOT salary > 30000
       ≡
salary <= 30000

(عكس ">" هو "<=" مش "<" بس!)
خطر شائع: لما تجمع AND و OR مع بعض في نفس الجملة، لازم تستخدم أقواس () توضّح ترتيب التنفيذ — زي المثال التاني، من غير الأقواس ممكن النتيجة تختلف تمامًا عن اللي قصدته.
03

Arithmetic Expressions — أعمدة محسوبة

تقدر تحسب قيمة جديدة جوه الـ SELECT نفسه من غير ما تغيّر البيانات الأصلية في الجدول — دايمًا اديها اسم بـ AS (alias).

new_column.sql
select fname , lname , salary ,
       salary + 1000 as new_salary
from employee;
bonus_raise.sql
-- بونص 10%
select fname , lname , salary ,
       salary * .1 as bonus from employee;

-- زيادة 15%
select fname , lname , salary ,
       salary * 1.15 as salary_after_raise
from employee;
مهم: العمليات الحسابية جوه SELECT بتحسب قيمة في الناتج بس — مش بتعدّل أي بيانات فعلية في الجدول. عايز تعدّل فعليًا؟ محتاج أمر UPDATE منفصل (زي ما اتعلمناه في الجلسة اللي فاتت).
04

ORDER BY — ترتيب النتائج

ASC تصاعدي (الافتراضي لو ماكتبتش حاجة)، DESC تنازلي. تقدر ترتب بأكتر من عمود مع بعض.

order_basic.sql
select fname , lname , salary from employee
order by salary desc;   -- من الأعلى للأقل
order_multi.sql
-- رتب بالمرتب، ولو اتساوى رتب بالاسم
select fname , lname , salary from employee
order by salary desc , fname asc;
لاحظ: تقدر تستخدم alias اتعمل جوه SELECT مباشرة في ORDER BY، زي: SELECT salary * 12 AS annual_salary ... ORDER BY annual_salary. العمود التاني في ORDER BY بيتفعّل بس لو فيه تعادل في العمود الأول.
05

INNER JOIN — ربط جدولين بشرط تطابق

لما البيانات اللي محتاجها موزّعة على جدولين، لازم تربطهم. INNER JOIN بيرجّع بس الصفوف اللي فيها تطابق في الجدولين.

inner_join.sql
select fname , lname , dname
from employee inner join department
on employee.dno = department.dnumber;
join_with_alias.sql
select concat(e.fname , ' ' , e.lname) as name,
       d.dname , e.salary
from employee e inner join department d
on e.dno = d.dnumber
where d.dname ilike 'Administration';
▸ العلاقة بينهم
employee.dno
     |
     V
department.dnumber

(عمود في كل جدول بيشاور على التاني)
Ambiguous Column Names: لو الجدولين فيهم عمود بنفس الاسم (زي name أو id)، لازم تحدد اسم الجدول قبل العمود (e.fname بدل fname بس) — وإلا SQL مش هتعرف تقصد عمود مين. عمومًا يُفضّل دايمًا تسبق كل عمود باسم جدوله (أو alias بتاعه) لوضوح أكتر وأداء أحسن.
06

LEFT / RIGHT / FULL / CROSS JOIN

INNER JOIN بيسيب أي صف مالوش تطابق. لو عايز تحتفظ بالصفوف اللي مالهاش تطابق كمان، محتاج نوع تاني من الـ JOIN.

النوعبيرجّع إيه
LEFT JOINكل صفوف الجدول الشمال، حتى لو مفيش تطابق في اليمين
RIGHT JOINكل صفوف الجدول اليمين، حتى لو مفيش تطابق في الشمال
FULL JOINكل صفوف الجدولين، سواء فيه تطابق أو لأ
CROSS JOINكل التوليفات الممكنة (لو 10 موظفين × 3 أقسام = 30 صف)
left_join.sql
-- كل الموظفين، حتى لو مالوش قسم مسجّل
select concat(e.fname,' ',e.lname) as name,
       d.dname , e.salary
from employee e left join department d
on e.dno = d.dnumber;
▸ مثال حقيقي (RIGHT JOIN)
Employee       | Dept_ID | Department
Ahmed Ali      | 100     | IT
Mohamed Samir  | 200     | Marketing
Mona Selim     | 100     | IT
(NULL)         | 300     | HR

(قسم HR فاضل ظاهر رغم مفيهوش موظف)
خطة تفكير سريعة: "عايز كل الصفوف من الجدول ده حتى لو مالوش تطابق؟" → استخدم LEFT (لو هو الجدول الأول) أو RIGHT (لو هو التاني). "عايز كل الصفوف من الجدولين مع بعض؟" → FULL. "مش محتاج شرط تطابق أصلاً، عايز كل التوليفات؟" → CROSS.
07

Self Join + Table Alias

أحيانًا الجدول محتاج "يتربط بنفسه" — زي جدول موظفين فيه عمود superssn بيشاور على ssn لموظف تاني في نفس الجدول (المشرف بتاعه).

self_join.sql
select e.name as employee_name,
       s.name as supervisor
from employees e , employees s
where e.supervisorID = s.ID;
▸ بيانات حقيقية من المحاضرة
ID   | Employee_name | SupervisorID
1122 | Ahmed Ali      | 2233
2233 | Kamel Mohamed  | 1234
1234 | Hanaa Sobhy    | 2233
3216 | Amr Omran      | NULL
5124 | Edward Hanna   | 2233
6699 | Mariam Adel    | 5124
ليه لازم alias هنا بالتحديد؟ عمليًا انت بتستخدم نفس الجدول employees مرتين في نفس الاستعلام — من غير alias (زي e وs)، SQL مش هتعرف تفرّق بين "الموظف" و"المشرف" لإن الاسمين واحد. Amr Omran في المثال ليه SupervisorID = NULL، يعني هو مش تحت إشراف حد — ده هيتشال من نتيجة self join عادي (INNER)، محتاج LEFT JOIN لو عايزه يفضل ظاهر.
08

Subqueries — استعلام جوه استعلام

الـ Subquery هي SELECT كاملة جوه SELECT تانية. الاستعلام الداخلي بينفّذ الأول، وناتجه بيستخدمه الاستعلام الخارجي.

subquery_max.sql
-- أعلى قسم مرتّب (استعلام جوه استعلام جوه استعلام)
select dname
from dept
where deptno = (
    select deptno from emp
    where sal = (select max(sal) from emp)
);
subquery_in.sql
-- موظفين شغالين في جيزة
select name
from Employee
where Dno in (
    select Dnumber from dept
    where location = 'giza'
);
خطوتين في التفكير: استعلام "أعلى راتب" مثلاً بيتحل على مرحلتين: (1) دوّر على أعلى قيمة (MAX(sal))، (2) دوّر على مين صاحبها. الـ Subquery بتخليك تكتب المرحلتين في جملة واحدة بدل ما تنفّذهم يدويًا.
09

UNION — دمج نتيجتين في واحدة

UNION بتدمج ناتج جملتين SELECT في نتيجة واحدة — بشرط: نفس عدد الأعمدة، ونفس نوع البيانات تقريبًا.

union_example.sql
select dnumber from department
where MRGSSN = 10
union
select dnumber from dept_locations
where dlocation = 'GIZA';
▸ UNION مقابل UNION ALL
UNION
  → بتشيل التكرار، كل قيمة تظهر مرة واحدة

UNION ALL
  → نفس الدمج، بس محتفظة بالتكرار
  → أسرع لأنها مش بتقارن الصفوف ببعض
مين بيسمّي الأعمدة في النتيجة النهائية؟ أسماء الأعمدة في النتيجة بتاخد اسمها من أول جملة SELECT بس، حتى لو الجملة التانية سمّت الأعمدة بشكل مختلف.
10

Aggregate Functions — دوال تجميعية

بتاخد مجموعة صفوف وترجّع نتيجة واحدة بس — زي المجموع أو المتوسط أو العدد. أشهرها: COUNT، SUM، AVG، MAX، MIN.

aggregate_basic.sql
select count(*) , max(salary) , min(salary),
       sum(salary) , avg(salary)
from employee;
count_with_join.sql
-- عدد موظفين قسم Research
select count(*)
from employee , department
where dno = dnumber and dname = 'Research';
▸ خطر شائع
الدوال التجميعية بتتجاهل قيم NULL
تلقائيًا في حساباتها — من غير ما
تحتاج تعمل أي فلترة إضافية بنفسك.
11

GROUP BY + HAVING

لو عايز نتيجة تجميعية لكل مجموعة بدل الجدول كله (زي متوسط راتب كل قسم لوحده)، استخدم GROUP BY. ولو عايز تفلتر المجموعات نفسها (مش الصفوف)، استخدم HAVING.

group_by.sql
select dname , count(*)
from employee e join department d
on e.dno = d.dnumber
group by dname;
having.sql
-- أقسام أكتر من 2 موظف فيها
select pnumber, pname, count(works_on.pno)
from project , Works_on
where Pnumber = Pno
group by pnumber, pname
having count(*) > 2
order by Pnumber;
الجزءبيفلتر إيهترتيب التنفيذ
WHEREصفوف مفردة، قبل التجميع1
GROUP BYبيكوّن المجموعات نفسها2
HAVINGالمجموعات، بعد ما اتكوّنت3
خطر شائع جدًا: WHERE مش ممكن تستخدم فيها دالة تجميعية زي COUNT() أو AVG() مباشرة — لازم HAVING. القاعدة: لو الشرط بيتكلم عن "صف واحد"، استخدم WHERE. لو بيتكلم عن "المجموعة ككل" (زي عدد أعضائها أو متوسطها)، استخدم HAVING.
12

NULL / LIMIT + خارطة التفكير السريعة

آخر شوية أدوات مهمة، وخلاصة تساعدك تقرر تكتب إيه في أي استعلام من أول وهلة.

null_check.sql
-- موظفين من غير مشرف
select fname, lname, superssn
from employee
where superssn is null;

-- موظفين ليهم مشرف
select fname, lname, superssn
from employee
where superssn is not null;
limit.sql
-- أعلى 5 رواتب
select fname, lname, salary
from employee
order by salary desc
limit 5;
NULL مش زي أي حاجة تانية: NULL معناها "مش معروف/مش موجود"، مش صفر ولا نص فاضي ولا false. أبدًا متكتبش column = NULL — ده مش هيشتغل أصلاً. لازم IS NULL أو IS NOT NULL.
